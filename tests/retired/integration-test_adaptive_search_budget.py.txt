async def new_search(app, *, token_budget=98305, request_budget=8, check_output_tokens=32768):
    client = httpx.AsyncClient(transport=httpx.ASGITransport(app=app), base_url="http://test",
        headers={"Authorization": "Bearer human"})

    async def post(path, payload, method="POST"):
        response = await client.request(method, path, json=payload,
            headers={"Authorization": "Bearer human", "Idempotency-Key": str(uuid4())})
        assert response.is_success, response.text
        return response.json()

    project = await post("/projects", {"title": "adaptive", "body": "Prove a fact."})
    await post(f"/projects/{project['project_id']}/runtime-settings", {
        "allow_real_api": True, "allowed_providers": ["deepseek"], "request_budget": 100}, "PUT")
    run = await post("/runs", {"branch_id": project["branch_id"], "goal_object_id": project["object_id"],
        "provider": "deepseek", "autonomous": True, "request_budget": request_budget,
        "max_output_tokens": 32768, "cumulative_output_token_budget": token_budget,
        "solver_controller": "bounded_search_v1", "search_config": {
            "budget_policy": "adaptive", "analysis_output_tokens": 8192,
            "min_work_output_tokens": 16384, "min_check_output_tokens": 16384,
            "final_output_tokens": 8192, "check_output_tokens": check_output_tokens,
            "check_requests": 2, "final_requests": 1,
        }})
    return client, run["run_id"]


def test_adaptive_root_ledger_counts_unknown_and_queued_holds(app):
    async def scenario():
        client, root_id = await new_search(app)
        try:
            runtime = Runtime(app.state.service)
            with app.state.database.sessions.begin() as session:
                search = runtime.search.session_for(session, root_id)
                initial = runtime.search.work_for(session, root_id)
                root = session.get(Run, root_id)
                attempt = Attempt(run_id=root_id, number=99, state="running", read_set={}, control_epoch=0,
                    token="test", lease_until="2099-01-01T00:00:00+00:00",
                    checkpoint={"search_work_id": initial.id})
                session.add(attempt)
                session.flush()
                session.add(ProviderRequest(project_id=search.project_id, run_id=root_id, attempt_id=attempt.id,
                    provider="deepseek", state="unknown", output_token_reservation=8192))
                queued_run = Run(branch_id=root.branch_id, goal_object_id=root.goal_object_id, provider="deepseek")
                session.add(queued_run)
                session.flush()
                session.add(SearchWork(root_run_id=root_id, run_id=queued_run.id, kind="advance", state="queued",
                    budget_pool="explore", input_snapshot={"output_cap": 4000}))
                session.flush()
                budget = runtime.search.budget(session, search)
                assert {row["remaining_output_tokens"] for row in budget.values()} == {86113}
                assert {row["remaining_requests"] for row in budget.values()} == {6}
                assert all(row["shared_root_budget"] for row in budget.values())
                recovery = runtime.search.budget(session, search, exclude_work=initial.id)
                assert recovery["explore"]["remaining_output_tokens"] == 86113
                assert recovery["explore"]["remaining_requests"] == 6
        finally:
            await client.aclose()
    asyncio.run(scenario())


def test_adaptive_minima_and_floors_block_small_dispatch(app):
    async def scenario():
        # This is enough for a root to exist, but not enough for meaningful
        # analysis plus an independent check and final delivery.
        client, root_id = await new_search(app, token_budget=30000, check_output_tokens=16384)
        try:
            runtime = Runtime(app.state.service)
            with app.state.database.sessions.begin() as session:
                search = runtime.search.session_for(session, root_id)
                assert runtime.search.minimum_output_tokens(session, search, "analysis") == 8192
                assert runtime.search.minimum_output_tokens(session, search, "check") == 16384
                assert runtime.search._adaptive_floor(search, "advance") == 24576
                assert runtime.search._adaptive_request_floor(search, "advance") == 2
                assert runtime.search.work_for(session, root_id) is None
                assert search.phase == "terminated"
                assert session.get(Run, root_id).state == "budget_exhausted"
        finally:
            await client.aclose()
    asyncio.run(scenario())


def test_adaptive_continues_one_route_then_reviews_candidate(app):
    class ProgressSolver(Solver):
        async def generate(self, task):
            response = await super().generate(task)
            if task["search_context"]["kind"] == "advance" and not task["search_context"]["route"]["progress"].get("steps"):
                response["result"].update(body="A useful intermediate result, still incomplete.",
                                           actions=[op("report_progress", status="progress")])
            return response

    async def scenario():
        client, _, _, root = await setup(app, config={
            "budget_policy": "adaptive", "max_routes": 1, "active_routes": 1,
        }, policy="reviewed_answer")
        solver = ProgressSolver()
        worker = HTTPWorker(client, providers=["deepseek"], provider_factory=lambda _: solver)
        try:
            claimed = await worker._post("/worker/claim-next", {"providers": ["deepseek"]})
            await worker.execute(claimed["task"])
            first = await worker._post("/worker/claim-next", {"providers": ["deepseek"]})
            concurrent = await worker._post("/worker/claim-next", {"providers": ["deepseek"]})
            assert concurrent["task"] is None
            await worker.execute(first["task"])
            second = await worker._post("/worker/claim-next", {"providers": ["deepseek"]})
            assert first["task"]["search_context"]["route_id"] == second["task"]["search_context"]["route_id"]
            await worker.execute(second["task"])
            await worker.run(once=True)
            assert [t["search_context"]["kind"] for t in solver.tasks] == ["analysis", "advance", "advance", "check", "final"]
            with app.state.database.sessions() as session:
                assert session.get(Run, root["run_id"]).state == "completed"
        finally:
            await client.aclose()
    asyncio.run(scenario())


def test_adaptive_first_pass_gives_each_route_one_attempt_before_continuing(app):
    class ProgressSolver(Solver):
        async def generate(self, task):
            response = await super().generate(task)
            if task["search_context"]["kind"] == "advance":
                response["result"].update(body="A route-local result that remains incomplete.",
                                           actions=[op("report_progress", status="progress")])
            return response

    async def scenario():
        client, _, _, _ = await setup(app, config={
            "budget_policy": "adaptive", "max_routes": 3, "active_routes": 1,
            "max_route_steps": 2, "route_schedule": "first_pass",
        })
        solver = ProgressSolver()
        worker = HTTPWorker(client, providers=["deepseek"], provider_factory=lambda _: solver)
        try:
            for _ in range(4):
                await worker.run(once=True)
            advances = [task for task in solver.tasks if task["search_context"]["kind"] == "advance"]
            # ``Solver`` proposes two route cards. Each receives its first
            # advance before either one receives a second.
            assert [task["search_context"]["route"]["ordinal"] for task in advances[:2]] == [0, 1]
        finally:
            await client.aclose()

    asyncio.run(scenario())


@pytest.mark.parametrize("calls,tokens", [(1, 70000), (6, 1)])
def test_adaptive_retry_cannot_spend_delivery_reserves(app, calls, tokens):
    async def scenario():
        client, root_id = await new_search(app)
        try:
            runtime = Runtime(app.state.service)
            with app.state.database.sessions.begin() as session:
                search = runtime.search.session_for(session, root_id)
                work = runtime.search.work_for(session, root_id)
                attempt = Attempt(run_id=root_id, number=99, state="running", read_set={}, control_epoch=0,
                    token="test", lease_until="2099-01-01T00:00:00+00:00", checkpoint={"search_work_id": work.id})
                session.add(attempt)
                session.flush()
                for _ in range(calls):
                    session.add(ProviderRequest(project_id=search.project_id, run_id=root_id,
                        attempt_id=attempt.id, provider="deepseek", state="spent",
                        usage={"completion_tokens": tokens}, output_token_reservation=tokens))
                session.flush()
                with pytest.raises(DomainError) as rejected:
                    runtime.search.admit(session, session.get(Run, root_id), 8192)
                assert rejected.value.response["error"] == "search_pool_exhausted"
                assert runtime.search.budget(session, search)["explore"]["requests"] == calls
        finally:
            await client.aclose()
    asyncio.run(scenario())
